--1. Использование Data Protection API в ASP.NET Core для шифрования данных конфигурации в промышленном окружении

Data Protection API — это высокоуровневый API для защиты конфиденциальных данных, таких как файлы cookie, токены и ключи. 
Он автоматически управляет созданием, ротацией и хранением ключей, обеспечивая изоляцию между приложениями.

--1.1 Добавьте в проект пакет Microsoft.AspNetCore.DataProtection.
dotnet add package Microsoft.AspNetCore.DataProtection

--1.2 Зарегистрируйте систему защиты данных:
services.AddDataProtection();

--1.2.1 Выполните настройку параметров по необходимости.

services.AddDataProtection()
    	.PersistKeysToFileSystem(new DirectoryInfo(@"путь_к_каталогу"))    
    	.SetApplicationName("RM.WebApi")
	.UseCryptographicAlgorithms(new Microsoft.AspNetCore.DataProtection.AuthenticatedEncryption.AuthenticatedEncryptionSettings
	    {
		EncryptionAlgorithm = Microsoft.AspNetCore.DataProtection.AuthenticatedEncryption.EncryptionAlgorithm.AES_256_CBC,
		ValidationAlgorithm = Microsoft.AspNetCore.DataProtection.AuthenticatedEncryption.ValidationAlgorithm.HMACSHA256
	    });
	    
PersistKeysToFileSystem 	— указывает, где хранить ключи шифрования (на Ubuntu лучше использовать /etc/ или /var/).
SetApplicationName 		— задаёт имя приложения для изоляции ключей (важно, если у вас несколько сервисов).
UseCryptographicAlgorithms 	— задаёт алгоритмы шифрования (AES‑256 и HMAC‑SHA256 — надёжный стандарт).

--1.3 Создание сервиса для работы с шифрованием

/// <summary>
/// Сервис защиты данных через ASP.NET Core Data Protection API.
/// </summary>
public interface IDataProtectionService
{
    /// <summary>
    /// Шифрует текст.
    /// </summary>
    /// <param name="plainText">Текст для шифрования.</param>
    /// <returns>Зашифрованный текст в виде строки.</returns>
    string Encrypt(string plainText);

    /// <summary>
    /// Дешифрует текст.
    /// </summary>
    /// <param name="cipherText">Зашифрованный текст в виде строки.</param>
    /// <returns>Расшифрованный текст.</returns>
    string Decrypt(string cipherText);
}

/// <summary>
/// Реализация сервиса защиты данных через ASP.NET Core Data Protection API.
/// </summary>
public class DataProtectionService : IDataProtectionService
{
    private readonly IDataProtector _protector;

    /// <summary>
    /// Инициализирует экземпляр <see cref="DataProtectionService"/>.
    /// </summary>
    /// <param name="provider">
    /// Поставщик защиты данных, предоставляемый инфраструктурой ASP.NET Core.
    /// </param>
    /// <param name="purpose">
    /// Цель шифрования данных. Логическая группа данных, для которой создаётся отдельный ключ.
    /// </param>
    public DataProtectionService(IDataProtectionProvider provider, string purpose)
    {
        ArgumentNullException.ThrowIfNull(provider, nameof(provider));
        ArgumentException.ThrowIfNullOrWhiteSpace(purpose, nameof(purpose));

        _protector = provider.CreateProtector(purpose);
    }

    /// <inheritdoc/>
    public string Encrypt(string plainText) => _protector.Protect(plainText);

    /// <inheritdoc/>
    public string Decrypt(string cipherText) => _protector.Unprotect(cipherText);

}

//Пользовательский (простой) тип, обязательно реализующий метод TryParse
public readonly struct ProductId
{
    public string Value { get; }

    private ProductId(string value) => Value = value;

    // Ключевой метод: его ищет инфраструктура привязки модели
    public static bool TryParse(string? s, out ProductId result)
    {
        result = default;
        if (string.IsNullOrEmpty(s)) return false;

        // Формат: буква 'p' + целое число, например p123
        if (s.Length < 2 || s[0] != 'p') return false;

        if (!int.TryParse(s.AsSpan(1), out _)) return false;

        result = new ProductId(s);
        return true;
    }
}

//Ограничение маршрута:
/// <summary>
/// Ограничение маршрута, проверяющее, что значение параметра
/// состоит из заданного строкового префикса и следующего за ним целого числа.
/// Например, при префиксе "p" принимаются значения "p123", "p456" и т. д.
/// </summary>
public class PrefixIdRouteConstraint : IRouteConstraint
{
    private readonly string _prefix;

    /// <summary>
    /// Инициализирует экземпляр <see cref="PrefixIdRouteConstraint"/>.
    /// </summary>
    /// <param name="prefix">Строковый префикс, с которого должно начинаться значение параметра маршрута.</param>
    public PrefixIdRouteConstraint(string prefix)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(prefix);

        _prefix = prefix;
    }

    /// <summary>
    /// Определяет, удовлетворяет ли значение параметра маршрута формату
    /// «префикс + целое число». Вызывается инфраструктурой маршрутизации
    /// как для входящих запросов, так и при генерации URL.
    /// </summary>
    /// <param name="httpContext">Текущий HTTP-контекст запроса (может быть null при генерации URL).</param>
    /// <param name="route">Маршрут, для которого проверяется ограничение.</param>
    /// <param name="routeKey">Имя параметра маршрута, к которому привязано ограничение (например, "id").</param>
    /// <param name="values">Словарь значений параметров маршрута, извлечённых из URL или переданных при генерации.</param>
    /// <param name="routeDirection">Направление маршрутизации: входящий запрос или генерация URL.</param>
    /// <returns>
    /// <see langword="true"/>, если значение существует, начинается с префикса
    /// и остаток строки является целым числом; иначе <see langword="false"/>.
    /// </returns>
    public bool Match(
        HttpContext? httpContext,
        IRouter? route,
        string routeKey,
        RouteValueDictionary values,
        RouteDirection routeDirection)
    {
        // 1. Извлекаем значение параметра маршрута по ключу routeKey.
        //    Если параметр отсутствует или равен null — ограничение не проходит.
        if (!values.TryGetValue(routeKey, out var value) || value is null)
            return false;

        // 2. Приводим значение к строке: в словаре оно может храниться
        //    как строка, число или другой объект.
        var str = value.ToString();

        // 3. Проверяем формат: строка длиннее префикса (значит, после префикса
        //    есть хотя бы один символ), начинается с префикса и остаток
        //    строки является целым числом. AsSpan избегает выделения
        //    промежуточной строки при проверке числовой части.
        return str?.Length > _prefix.Length
            && str.StartsWith(_prefix, StringComparison.Ordinal)
            && int.TryParse(str.AsSpan(_prefix.Length), out _);
    }
}

{
    options.ConstraintMap.Add(nameof(ProductId), typeof(PrefixIdRouteConstraint));
});
